cmake_minimum_required(VERSION 3.20)
project(CPAMPMonitor VERSION 1.0.1 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "Build on Windows with MSVC (TrafficMonitor uses the MSVC C++ ABI)")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/utf-8 /W4 /wd4100)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_library(cpamp_core STATIC src/quota.cpp src/platform.cpp)
target_include_directories(cpamp_core PUBLIC src third_party)
target_link_libraries(cpamp_core PUBLIC winhttp crypt32 shell32 ole32 user32)
add_library(CPAMPMonitor SHARED src/plugin.cpp src/plugin.rc)
target_include_directories(CPAMPMonitor PRIVATE include)
target_link_libraries(CPAMPMonitor PRIVATE cpamp_core gdi32)
add_executable(CPAMPTests tests/tests.cpp)
target_link_libraries(CPAMPTests PRIVATE cpamp_core)
add_executable(PluginProbe tests/plugin_probe.cpp)
target_include_directories(PluginProbe PRIVATE include)
target_link_libraries(PluginProbe PRIVATE user32 gdi32)
enable_testing()
add_test(NAME Core COMMAND CPAMPTests)
add_test(NAME PluginABI COMMAND PluginProbe $<TARGET_FILE:CPAMPMonitor>)
